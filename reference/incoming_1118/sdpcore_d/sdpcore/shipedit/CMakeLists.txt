# Редактор данных судна (Qt 6: Widgets, Network). Собирается, если в корневом CMake включено SDP_BUILD_EDITOR.
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
set(CMAKE_AUTOMOC ON)
qt_add_executable(sdp_ship main.cpp mainwindow.cpp mainwindow.h viewedit.cpp viewedit.h keystore.cpp keystore.h)
target_link_libraries(sdp_ship PRIVATE dpcore Qt6::Widgets Qt6::Network)
set_target_properties(sdp_ship PROPERTIES MACOSX_BUNDLE OFF WIN32_EXECUTABLE ON)
# ключ API — в защищённом хранилище системы (shipedit/keystore.cpp)
if(APPLE)
  target_link_libraries(sdp_ship PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(sdp_ship PRIVATE advapi32)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIBSECRET QUIET IMPORTED_TARGET libsecret-1)
  endif()
  if(LIBSECRET_FOUND)
    target_link_libraries(sdp_ship PRIVATE PkgConfig::LIBSECRET)
    target_compile_definitions(sdp_ship PRIVATE SDP_HAVE_LIBSECRET)
  else()
    message(STATUS "libsecret не найден: ключ API в редакторе будет храниться только до закрытия программы")
  endif()
endif()
